// FORM = 50
// CAPITOL = 1061
// Rânduri statice
// Cap.IV: Col.1 (Rând.*) = Col.2 + Col.5 + Col.7 + Col.8

function calcCap1061Col1() {
    var rows = [
        { id: "119153", rind: "010" },
        { id: "119154", rind: "020" },
        { id: "119152", rind: "030" }
    ];

    rows.forEach(function (row) {
        var col2 = parseInt(
            $("#50_1061_" + row.id + "_" + row.rind + "_2").val(),
            10
        ) || 0;

        var col5 = parseInt(
            $("#50_1061_" + row.id + "_" + row.rind + "_5").val(),
            10
        ) || 0;

        var col7 = parseInt(
            $("#50_1061_" + row.id + "_" + row.rind + "_7").val(),
            10
        ) || 0;

        var col8 = parseInt(
            $("#50_1061_" + row.id + "_" + row.rind + "_8").val(),
            10
        ) || 0;

        var sum = col2 + col5 + col7 + col8;

        $("#50_1061_" + row.id + "_" + row.rind + "_1")
            .val(sum !== 0 ? sum : "")
            .prop("readonly", true);
    });
}

function f_Capitol_1061() {
    calcCap1061Col1();
}

$(document).ready(function () {
    f_Capitol_1061();

    $("input:not([type='button']):not([readonly]):not([disabled])")
        .on("change", f_Capitol_1061);

    $(document).on(
        "change",
        "input:not([type='button']):not([readonly]):not([disabled])",
        f_Capitol_1061
    );
});
